CREATE TABLE Oddzialy (
  ID_Oddzialu  NUMBER PRIMARY KEY,
  Nazwa        VARCHAR2(100) NOT NULL,
  Miasto       VARCHAR2(50)  NOT NULL,
  Adres        VARCHAR2(100) NOT NULL
);

CREATE TABLE Modele_Konfiguracje (
  ID_Modelu    NUMBER PRIMARY KEY,
  ID_Kategorii NUMBER NOT NULL,
  Marka        VARCHAR2(50) NOT NULL,
  Model        VARCHAR2(50) NOT NULL,
  Silnik       VARCHAR2(50),
  Sila         VARCHAR2(20),
  CONSTRAINT fk_model_kategoria 
    FOREIGN KEY (ID_Kategorii) REFERENCES Dict_Kategorie(ID_Kategorii)
);

CREATE TABLE Cenniki (
  ID_Cennika    NUMBER PRIMARY KEY,
  ID_Kategorii  NUMBER NOT NULL,
  Cena_Doba     NUMBER(10,2) NOT NULL,
  Obowiazuje_Od DATE NOT NULL,
  Obowiazuje_Do DATE,
  CONSTRAINT fk_cennik_kategoria 
    FOREIGN KEY (ID_Kategorii) REFERENCES Dict_Kategorie(ID_Kategorii)
);

CREATE TABLE Pracownicy (
  ID_Pracownika NUMBER PRIMARY KEY,
  ID_Oddzialu   NUMBER NOT NULL,
  Imie          VARCHAR2(50) NOT NULL,
  Nazwisko      VARCHAR2(50) NOT NULL,
  PESEL         CHAR(11) UNIQUE,
  Rola          VARCHAR2(30) NOT NULL,
  CONSTRAINT chk_pracownik_rola 
    CHECK (Rola IN ('Administrator','Agent','Ksiegowy')),
  CONSTRAINT fk_pracownik_oddzial 
    FOREIGN KEY (ID_Oddzialu) REFERENCES Oddzialy(ID_Oddzialu)
);

CREATE TABLE Samochody (
  ID_Samochodu      NUMBER PRIMARY KEY,
  ID_Modelu         NUMBER NOT NULL,
  ID_Statusu        NUMBER NOT NULL,
  ID_Oddzialu       NUMBER NOT NULL,
  Nr_Rejestracyjny  VARCHAR2(10) UNIQUE NOT NULL,
  VIN               CHAR(17) UNIQUE NOT NULL,
  Przebieg          NUMBER DEFAULT 0,
  CONSTRAINT fk_samochod_model 
    FOREIGN KEY (ID_Modelu) REFERENCES Modele_Konfiguracje(ID_Modelu),
  CONSTRAINT fk_samochod_status 
    FOREIGN KEY (ID_Statusu) REFERENCES Dict_Statusy_Pojazdu(ID_Statusu),
  CONSTRAINT fk_samochod_oddzial 
    FOREIGN KEY (ID_Oddzialu) REFERENCES Oddzialy(ID_Oddzialu)
);
/////////////////
CREATE TABLE Klienci (
  ID_Klienta  NUMBER PRIMARY KEY,
  Imie        VARCHAR2(50)  NOT NULL,
  Nazwisko    VARCHAR2(50)  NOT NULL,
  PESEL       CHAR(11)      UNIQUE,
  NIP         CHAR(10),
  Email       VARCHAR2(100) UNIQUE NOT NULL,
  Telefon     VARCHAR2(15)
);

CREATE TABLE Wypozyczenia (
  ID_Wypozyczenia           NUMBER PRIMARY KEY,
  ID_Klienta                NUMBER NOT NULL,
  ID_Samochodu              NUMBER NOT NULL,
  ID_Pracownika_Wydajacego  NUMBER,
  ID_Pracownika_Odbierajacego NUMBER,
  Data_Wypozyczenia         TIMESTAMP,
  Data_Zwrotu_Planowana     DATE NOT NULL,
  Data_Zwrotu_Rzeczywista   TIMESTAMP,
  Status                    VARCHAR2(20) NOT NULL,
  CONSTRAINT chk_wyp_status 
    CHECK (Status IN ('Rezerwacja','Aktywne','Zakonczone','Anulowane')),
  CONSTRAINT fk_wyp_klient 
    FOREIGN KEY (ID_Klienta) REFERENCES Klienci(ID_Klienta),
  CONSTRAINT fk_wyp_samochod 
    FOREIGN KEY (ID_Samochodu) REFERENCES Samochody(ID_Samochodu),
  CONSTRAINT fk_wyp_pracownik_wyd 
    FOREIGN KEY (ID_Pracownika_Wydajacego) REFERENCES Pracownicy(ID_Pracownika),
  CONSTRAINT fk_wyp_pracownik_odb 
    FOREIGN KEY (ID_Pracownika_Odbierajacego) REFERENCES Pracownicy(ID_Pracownika)
);

CREATE TABLE Faktury (
  ID_Faktury       NUMBER PRIMARY KEY,
  ID_Wypozyczenia  NUMBER UNIQUE NOT NULL,
  Data_Wystawienia DATE NOT NULL,
  Kwota_Netto      NUMBER(10,2),
  Kwota_Brutto     NUMBER(10,2),
  Status_Platnosci VARCHAR2(20) NOT NULL,
  CONSTRAINT chk_faktura_status 
    CHECK (Status_Platnosci IN ('Oczekuje','Oplacona','Zalegla')),
  CONSTRAINT fk_faktura_wyp 
    FOREIGN KEY (ID_Wypozyczenia) REFERENCES Wypozyczenia(ID_Wypozyczenia)
);

CREATE TABLE Historia_Logi_Pojazdu (
  ID_Logu       NUMBER PRIMARY KEY,
  ID_Samochodu  NUMBER NOT NULL,
  ID_Pracownika NUMBER,
  Data_Zmiany   TIMESTAMP DEFAULT SYSTIMESTAMP,
  Stary_Status  NUMBER,
  Nowy_Status   NUMBER NOT NULL,
  CONSTRAINT fk_log_samochod 
    FOREIGN KEY (ID_Samochodu) REFERENCES Samochody(ID_Samochodu),
  CONSTRAINT fk_log_pracownik 
    FOREIGN KEY (ID_Pracownika) REFERENCES Pracownicy(ID_Pracownika),
  CONSTRAINT fk_log_stary_status 
    FOREIGN KEY (Stary_Status) REFERENCES Dict_Statusy_Pojazdu(ID_Statusu),
  CONSTRAINT fk_log_nowy_status 
    FOREIGN KEY (Nowy_Status) REFERENCES Dict_Statusy_Pojazdu(ID_Statusu)
);

/////////////////Sekwencje w Oracle to automatyczne liczniki — zastępują AUTO_INCREMENT//////////////////////////////
//////////////WYZWALACZEodpalają się sami przed każdym INSERT.
Działają tak:
Ty piszesz:
INSERT INTO Klienci(Imie, Nazwisko, Email) VALUES('Jan','Kowalski','jan@wp.pl');

Oracle przed wykonaniem INSERT odpala trigger i uzupełnia:
ID_Klienta = seq_klienci.NEXTVAL  → np. 1

Wynik w bazie:
ID=1, Imie='Jan', Nazwisko='Kowalski', Email='jan@wp.pl'///////////////////////////

CREATE OR REPLACE TRIGGER trg_oddzialy_bi
BEFORE INSERT ON Oddzialy FOR EACH ROW
BEGIN
  IF :NEW.ID_Oddzialu IS NULL THEN
    :NEW.ID_Oddzialu := seq_oddzialy.NEXTVAL;
  END IF;
END;
/

CREATE OR REPLACE TRIGGER trg_status_pojazdu_au
AFTER UPDATE OF ID_Statusu ON Samochody FOR EACH ROW
BEGIN
  INSERT INTO Historia_Logi_Pojazdu (
    ID_Logu,
    ID_Samochodu,
    ID_Pracownika,
    Data_Zmiany,
    Stary_Status,
    Nowy_Status
  ) VALUES (
    seq_logi.NEXTVAL,
    :OLD.ID_Samochodu,
    NULL,
    SYSTIMESTAMP,
    :OLD.ID_Statusu,
    :NEW.ID_Statusu
  );
END;
/
