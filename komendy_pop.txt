-- Tabele słownikowe (bez zależności)
CREATE TABLE Dict_Kategorie (
  ID_Kategorii  NUMBER PRIMARY KEY,
  Nazwa_Kategorii VARCHAR2(30) NOT NULL
);

CREATE TABLE Dict_Statusy_Pojazdu (
  ID_Statusu    NUMBER PRIMARY KEY,
  Nazwa_Statusu VARCHAR2(30) NOT NULL
);

-- Tabele główne
CREATE TABLE Oddzialy (
  ID_Oddzialu  NUMBER PRIMARY KEY,
  Nazwa        VARCHAR2(100) NOT NULL,
  Miasto       VARCHAR2(50)  NOT NULL,
  Adres        VARCHAR2(100) NOT NULL
);

CREATE TABLE Modele_Konfiguracje (
  ID_Modelu    NUMBER PRIMARY KEY,
  ID_Kategorii NUMBER NOT NULL,
  Marka        VARCHAR2(50) NOT NULL,
  Model        VARCHAR2(50) NOT NULL,
  Silnik       VARCHAR2(50),
  Sila         VARCHAR2(20),
  CONSTRAINT fk_model_kategoria 
    FOREIGN KEY (ID_Kategorii) REFERENCES Dict_Kategorie(ID_Kategorii)
);

CREATE TABLE Cenniki (
  ID_Cennika    NUMBER PRIMARY KEY,
  ID_Kategorii  NUMBER NOT NULL,
  Cena_Doba     NUMBER(10,2) NOT NULL,
  Obowiazuje_Od DATE NOT NULL,
  Obowiazuje_Do DATE,
  CONSTRAINT fk_cennik_kategoria 
    FOREIGN KEY (ID_Kategorii) REFERENCES Dict_Kategorie(ID_Kategorii)
);

CREATE TABLE Pracownicy (
  ID_Pracownika NUMBER PRIMARY KEY,
  ID_Oddzialu   NUMBER NOT NULL,
  Imie          VARCHAR2(50) NOT NULL,
  Nazwisko      VARCHAR2(50) NOT NULL,
  PESEL         CHAR(11) UNIQUE,
  Rola          VARCHAR2(30) NOT NULL,
  CONSTRAINT chk_pracownik_rola 
    CHECK (Rola IN ('Administrator','Agent','Ksiegowy')),
  CONSTRAINT fk_pracownik_oddzial 
    FOREIGN KEY (ID_Oddzialu) REFERENCES Oddzialy(ID_Oddzialu)
);

CREATE TABLE Samochody (
  ID_Samochodu      NUMBER PRIMARY KEY,
  ID_Modelu         NUMBER NOT NULL,
  ID_Statusu        NUMBER NOT NULL,
  ID_Oddzialu       NUMBER NOT NULL,
  Nr_Rejestracyjny  VARCHAR2(10) UNIQUE NOT NULL,
  VIN               CHAR(17) UNIQUE NOT NULL,
  Przebieg          NUMBER DEFAULT 0,
  CONSTRAINT fk_samochod_model 
    FOREIGN KEY (ID_Modelu) REFERENCES Modele_Konfiguracje(ID_Modelu),
  CONSTRAINT fk_samochod_status 
    FOREIGN KEY (ID_Statusu) REFERENCES Dict_Statusy_Pojazdu(ID_Statusu),
  CONSTRAINT fk_samochod_oddzial 
    FOREIGN KEY (ID_Oddzialu) REFERENCES Oddzialy(ID_Oddzialu)
);

CREATE TABLE Klienci (
  ID_Klienta  NUMBER PRIMARY KEY,
  Imie        VARCHAR2(50)  NOT NULL,
  Nazwisko    VARCHAR2(50)  NOT NULL,
  PESEL       CHAR(11)      UNIQUE,
  NIP         CHAR(10),
  Email       VARCHAR2(100) UNIQUE NOT NULL,
  Telefon     VARCHAR2(15)
);

CREATE TABLE Wypozyczenia (
  ID_Wypozyczenia           NUMBER PRIMARY KEY,
  ID_Klienta                NUMBER NOT NULL,
  ID_Samochodu              NUMBER NOT NULL,
  ID_Pracownika_Wydajacego  NUMBER,
  ID_Pracownika_Odbierajacego NUMBER,
  Data_Wypozyczenia         TIMESTAMP,
  Data_Zwrotu_Planowana     DATE NOT NULL,
  Data_Zwrotu_Rzeczywista   TIMESTAMP,
  Status                    VARCHAR2(20) NOT NULL,
  CONSTRAINT chk_wyp_status 
    CHECK (Status IN ('Rezerwacja','Aktywne','Zakonczone','Anulowane')),
  CONSTRAINT fk_wyp_klient 
    FOREIGN KEY (ID_Klienta) REFERENCES Klienci(ID_Klienta),
  CONSTRAINT fk_wyp_samochod 
    FOREIGN KEY (ID_Samochodu) REFERENCES Samochody(ID_Samochodu),
  CONSTRAINT fk_wyp_pracownik_wyd 
    FOREIGN KEY (ID_Pracownika_Wydajacego) REFERENCES Pracownicy(ID_Pracownika),
  CONSTRAINT fk_wyp_pracownik_odb 
    FOREIGN KEY (ID_Pracownika_Odbierajacego) REFERENCES Pracownicy(ID_Pracownika)
);

CREATE TABLE Faktury (
  ID_Faktury       NUMBER PRIMARY KEY,
  ID_Wypozyczenia  NUMBER UNIQUE NOT NULL,
  Data_Wystawienia DATE NOT NULL,
  Kwota_Netto      NUMBER(10,2),
  Kwota_Brutto     NUMBER(10,2),
  Status_Platnosci VARCHAR2(20) NOT NULL,
  CONSTRAINT chk_faktura_status 
    CHECK (Status_Platnosci IN ('Oczekuje','Oplacona','Zalegla')),
  CONSTRAINT fk_faktura_wyp 
    FOREIGN KEY (ID_Wypozyczenia) REFERENCES Wypozyczenia(ID_Wypozyczenia)
);

CREATE TABLE Historia_Logi_Pojazdu (
  ID_Logu       NUMBER PRIMARY KEY,
  ID_Samochodu  NUMBER NOT NULL,
  ID_Pracownika NUMBER,
  Data_Zmiany   TIMESTAMP DEFAULT SYSTIMESTAMP,
  Stary_Status  NUMBER,
  Nowy_Status   NUMBER NOT NULL,
  CONSTRAINT fk_log_samochod 
    FOREIGN KEY (ID_Samochodu) REFERENCES Samochody(ID_Samochodu),
  CONSTRAINT fk_log_pracownik 
    FOREIGN KEY (ID_Pracownika) REFERENCES Pracownicy(ID_Pracownika),
  CONSTRAINT fk_log_stary_status 
    FOREIGN KEY (Stary_Status) REFERENCES Dict_Statusy_Pojazdu(ID_Statusu),
  CONSTRAINT fk_log_nowy_status 
    FOREIGN KEY (Nowy_Status) REFERENCES Dict_Statusy_Pojazdu(ID_Statusu)
);

CREATE OR REPLACE PACKAGE pkg_wypozyczalnia AS
  PROCEDURE RezerwujPojazd(
    p_id_klienta   IN NUMBER,
    p_id_samochodu IN NUMBER,
    p_data_od      IN DATE,
    p_data_do      IN DATE,
    p_id_wyp       OUT NUMBER

    p_id_pracownika   IN NUMBER
  );
  PROCEDURE ZwrocPojazd(
    p_id_wypozyczenia IN NUMBER,
    p_id_pracownika   IN NUMBER,
    p_przebieg_konc   IN NUMBER
  );
  FUNCTION ObliczKoszt(
    p_id_wypozyczenia IN NUMBER
  ) RETURN NUMBER;
  PROCEDURE GenerujRaportMiesięczny(
    p_rok     IN NUMBER,
    p_miesiac IN NUMBER
  );
END pkg_wypozyczalnia;
/

CREATE OR REPLACE PACKAGE BODY pkg_wypozyczalnia AS

  PROCEDURE RezerwujPojazd(
    p_id_klienta   IN NUMBER,
    p_id_samochodu IN NUMBER,
    p_data_od      IN DATE,
    p_data_do      IN DATE,
    p_id_wyp       OUT NUMBER
  ) AS
    v_status_dostepny  NUMBER;
    v_status_samochodu NUMBER;
    v_check            NUMBER;
  BEGIN
    SELECT ID_Statusu INTO v_status_samochodu
    FROM Samochody WHERE ID_Samochodu = p_id_samochodu;

    SELECT ID_Statusu INTO v_status_dostepny
    FROM Dict_Statusy_Pojazdu WHERE Nazwa_Statusu = 'Dostepny';

    IF v_status_samochodu != v_status_dostepny THEN
      RAISE_APPLICATION_ERROR(-20001, 'Pojazd nie jest dostepny do rezerwacji.');
    END IF;

    SELECT COUNT(*) INTO v_check FROM Klienci WHERE ID_Klienta = p_id_klienta;
    IF v_check = 0 THEN
      RAISE_APPLICATION_ERROR(-20002, 'Klient o podanym ID nie istnieje.');
    END IF;

    -- UWAGA: Nie zmieniamy statusu samochodu przy rezerwacji – zmiana nastąpi dopiero przy wydaniu
    INSERT INTO Wypozyczenia(
      ID_Klienta, ID_Samochodu, Data_Zwrotu_Planowana, Status
    ) VALUES (
      p_id_klienta, p_id_samochodu, p_data_do, 'Rezerwacja'
    ) RETURNING ID_Wypozyczenia INTO p_id_wyp;

    COMMIT;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20003, 'Nie znaleziono pojazdu o podanym ID.');
    WHEN OTHERS THEN
      ROLLBACK; RAISE;
  END RezerwujPojazd;

  PROCEDURE WydajPojazd(
    p_id_wypozyczenia IN NUMBER,
    p_id_pracownika   IN NUMBER
  ) AS
    v_status       VARCHAR2(20);
    v_id_samochodu NUMBER;
  BEGIN
    SELECT Status, ID_Samochodu INTO v_status, v_id_samochodu
    FROM Wypozyczenia WHERE ID_Wypozyczenia = p_id_wypozyczenia;

    IF v_status != 'Rezerwacja' THEN
      RAISE_APPLICATION_ERROR(-20004,
        'Wypozyczenie nie ma statusu Rezerwacja. Aktualny status: ' || v_status);
    END IF;

    UPDATE Wypozyczenia SET
      Status = 'Aktywne',
      Data_Wypozyczenia = SYSTIMESTAMP,
      ID_Pracownika_Wydajacego = p_id_pracownika
    WHERE ID_Wypozyczenia = p_id_wypozyczenia;

    -- Zmiana statusu samochodu na 'Wynajety' (ID_Statusu = 2) dopiero przy wydaniu
    UPDATE Samochody SET ID_Statusu = 2
    WHERE ID_Samochodu = v_id_samochodu;

    COMMIT;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20005, 'Nie znaleziono wypozyczenia o podanym ID.');
    WHEN OTHERS THEN
      ROLLBACK; RAISE;
  END WydajPojazd;

  PROCEDURE ZwrocPojazd(
    p_id_wypozyczenia IN NUMBER,
    p_id_pracownika   IN NUMBER,
    p_przebieg_konc   IN NUMBER
  ) AS
    v_status       VARCHAR2(20);
    v_id_samochodu NUMBER;
    v_koszt_netto  NUMBER;
    v_koszt_brutto NUMBER;
  BEGIN
    SELECT Status, ID_Samochodu INTO v_status, v_id_samochodu
    FROM Wypozyczenia WHERE ID_Wypozyczenia = p_id_wypozyczenia;

    IF v_status != 'Aktywne' THEN
      RAISE_APPLICATION_ERROR(-20006,
        'Wypozyczenie nie ma statusu Aktywne. Aktualny status: ' || v_status);
    END IF;

    UPDATE Wypozyczenia SET
      Status = 'Zakonczone',
      Data_Zwrotu_Rzeczywista = SYSTIMESTAMP,
      ID_Pracownika_Odbierajacego = p_id_pracownika
    WHERE ID_Wypozyczenia = p_id_wypozyczenia;

    UPDATE Samochody SET
      Przebieg = p_przebieg_konc, ID_Statusu = 1
    WHERE ID_Samochodu = v_id_samochodu;

    v_koszt_netto  := ObliczKoszt(p_id_wypozyczenia);
    v_koszt_brutto := ROUND(v_koszt_netto * 1.23, 2);

    INSERT INTO Faktury(
      ID_Wypozyczenia, Data_Wystawienia, Kwota_Netto, Kwota_Brutto, Status_Platnosci
    ) VALUES (
      p_id_wypozyczenia, SYSDATE, v_koszt_netto, v_koszt_brutto, 'Oczekuje'
    );
    COMMIT;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20007, 'Nie znaleziono wypozyczenia o podanym ID.');
    WHEN OTHERS THEN
      ROLLBACK; RAISE;
  END ZwrocPojazd;

  FUNCTION ObliczKoszt(p_id_wypozyczenia IN NUMBER) RETURN NUMBER AS
    v_dni       NUMBER;
    v_cena_doba NUMBER;
  BEGIN
    SELECT
      CEIL(CAST(w.Data_Zwrotu_Rzeczywista AS DATE) -
           CAST(w.Data_Wypozyczenia AS DATE)),
      c.Cena_Doba
    INTO v_dni, v_cena_doba
    FROM Wypozyczenia w
    JOIN Samochody s ON w.ID_Samochodu = s.ID_Samochodu
    JOIN Modele_Konfiguracje m ON s.ID_Modelu = m.ID_Modelu
    JOIN Cenniki c ON m.ID_Kategorii = c.ID_Kategorii
    WHERE w.ID_Wypozyczenia = p_id_wypozyczenia
      AND (c.Obowiazuje_Do IS NULL OR c.Obowiazuje_Do >= SYSDATE);
    RETURN v_dni * v_cena_doba;
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20008, 'Nie znaleziono danych do obliczenia kosztu.');
  END ObliczKoszt;

  PROCEDURE GenerujRaportMiesięczny(
    p_rok     IN NUMBER,
    p_miesiac IN NUMBER
  ) AS
    v_liczba_wypozyczen     NUMBER;
    v_przychod_netto        NUMBER;
    v_przychod_brutto       NUMBER;
    v_liczba_niezaplaconych NUMBER;
    CURSOR c_niezaplacone IS
      SELECT f.ID_Faktury, k.Imie, k.Nazwisko, k.Email,
             f.Kwota_Brutto, f.Data_Wystawienia
      FROM Faktury f
      JOIN Wypozyczenia w ON f.ID_Wypozyczenia = w.ID_Wypozyczenia
      JOIN Klienci k ON w.ID_Klienta = k.ID_Klienta
      WHERE f.Status_Platnosci != 'Oplacona'
        AND EXTRACT(YEAR  FROM f.Data_Wystawienia) = p_rok
        AND EXTRACT(MONTH FROM f.Data_Wystawienia) = p_miesiac;
  BEGIN
    SELECT COUNT(*), NVL(SUM(f.Kwota_Netto),0), NVL(SUM(f.Kwota_Brutto),0)
    INTO v_liczba_wypozyczen, v_przychod_netto, v_przychod_brutto
    FROM Wypozyczenia w
    JOIN Faktury f ON w.ID_Wypozyczenia = f.ID_Wypozyczenia
    WHERE EXTRACT(YEAR  FROM w.Data_Zwrotu_Rzeczywista) = p_rok
      AND EXTRACT(MONTH FROM w.Data_Zwrotu_Rzeczywista) = p_miesiac
      AND w.Status = 'Zakonczone';

    DBMS_OUTPUT.PUT_LINE('========================================');
    DBMS_OUTPUT.PUT_LINE('RAPORT MIESIĘCZNY: ' || p_miesiac || '/' || p_rok);
    DBMS_OUTPUT.PUT_LINE('========================================');
    DBMS_OUTPUT.PUT_LINE('Liczba wypożyczeń: ' || v_liczba_wypozyczen);
    DBMS_OUTPUT.PUT_LINE('Przychód netto:    ' || v_przychod_netto  || ' PLN');
    DBMS_OUTPUT.PUT_LINE('Przychód brutto:   ' || v_przychod_brutto || ' PLN');
    DBMS_OUTPUT.PUT_LINE('----------------------------------------');
    DBMS_OUTPUT.PUT_LINE('NIEZAPŁACONE FAKTURY:');

    v_liczba_niezaplaconych := 0;
    FOR rec IN c_niezaplacone LOOP
      v_liczba_niezaplaconych := v_liczba_niezaplaconych + 1;
      DBMS_OUTPUT.PUT_LINE(
        'Faktura #' || rec.ID_Faktury ||
        ' | ' || rec.Imie || ' ' || rec.Nazwisko ||
        ' | ' || rec.Kwota_Brutto || ' PLN' ||
        ' | ' || TO_CHAR(rec.Data_Wystawienia, 'DD-MM-YYYY')
      );
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Niezapłaconych: ' || v_liczba_niezaplaconych);
    DBMS_OUTPUT.PUT_LINE('========================================');
  END GenerujRaportMiesięczny;

END pkg_wypozyczalnia;

-- Generator 500 zakończonych wypożyczeń z ostatnich 2 lat
BEGIN
  FOR i IN 1..500 LOOP
    DECLARE
      v_id_wyp     NUMBER;
      v_id_klienta NUMBER;
      v_id_sam     NUMBER;
      v_data_od    TIMESTAMP;
      v_data_do    DATE;
      v_data_zwrot TIMESTAMP;
      v_dni        NUMBER;
      v_koszt      NUMBER;
      v_cena       NUMBER;
    BEGIN
      v_id_klienta := MOD(i, 13) + 1;
      v_id_sam     := MOD(i, 14) + 1;
      v_data_od    := SYSTIMESTAMP - (MOD(i * 7, 730) + 1);
      v_dni        := MOD(i, 14) + 1;
      v_data_do    := CAST(v_data_od AS DATE) + v_dni;
      v_data_zwrot := v_data_od + v_dni;

      INSERT INTO Wypozyczenia(
        ID_Klienta, ID_Samochodu,
        Data_Wypozyczenia, Data_Zwrotu_Planowana,
        Data_Zwrotu_Rzeczywista,
        ID_Pracownika_Wydajacego, ID_Pracownika_Odbierajacego,
        Status
      ) VALUES (
        v_id_klienta, v_id_sam,
        v_data_od, v_data_do,
        v_data_zwrot,
        MOD(i, 3) + 1, MOD(i, 3) + 1,
        'Zakonczone'
      ) RETURNING ID_Wypozyczenia INTO v_id_wyp;

      SELECT c.Cena_Doba INTO v_cena
      FROM Cenniki c
      JOIN Modele_Konfiguracje m ON c.ID_Kategorii = m.ID_Kategorii
      JOIN Samochody s ON m.ID_Modelu = s.ID_Modelu
      WHERE s.ID_Samochodu = v_id_sam
        AND (c.Obowiazuje_Do IS NULL OR c.Obowiazuje_Do >= SYSDATE)
        AND ROWNUM = 1;

      v_koszt := v_dni * v_cena;

      INSERT INTO Faktury(
        ID_Wypozyczenia, Data_Wystawienia,
        Kwota_Netto, Kwota_Brutto, Status_Platnosci
      ) VALUES (
        v_id_wyp, CAST(v_data_zwrot AS DATE),
        v_koszt, ROUND(v_koszt * 1.23, 2),
        CASE WHEN MOD(i, 10) = 0 THEN 'Zalegla'
             WHEN MOD(i, 5)  = 0 THEN 'Oczekuje'
             ELSE 'Oplacona' END
      );

      -- Zmiana statusu (trigger loguje)
      UPDATE Samochody SET ID_Statusu = 2 WHERE ID_Samochodu = v_id_sam;
      UPDATE Samochody SET ID_Statusu = 1 WHERE ID_Samochodu = v_id_sam;

    EXCEPTION
      WHEN OTHERS THEN NULL;
    END;
  END LOOP;
  COMMIT;
END;

CREATE TABLE Wypozyczenia_Archiwum AS 
SELECT w.*, CAST(NULL AS TIMESTAMP) AS Data_Archiwizacji
FROM Wypozyczenia w WHERE 1=0;

ALTER TABLE Wypozyczenia_Archiwum MODIFY Data_Archiwizacji DEFAULT SYSTIMESTAMP;

CREATE INDEX idx_arch_data ON Wypozyczenia_Archiwum(Data_Zwrotu_Rzeczywista);
CREATE INDEX idx_arch_klient ON Wypozyczenia_Archiwum(ID_Klienta);

CREATE OR REPLACE PROCEDURE ArchiwizujWypozyczenia(
  p_miesiac_wstecz IN NUMBER DEFAULT 12
) AS
  v_liczba_przeniesonych NUMBER := 0;
  v_data_graniczna       DATE;
BEGIN
  v_data_graniczna := ADD_MONTHS(SYSDATE, -p_miesiac_wstecz);

  INSERT INTO Wypozyczenia_Archiwum (ID_Wypozyczenia, ID_Klienta, ID_Samochodu,
    ID_Pracownika_Wydajacego, ID_Pracownika_Odbierajacego,
    Data_Wypozyczenia, Data_Zwrotu_Planowana, Data_Zwrotu_Rzeczywista, Status,
    Data_Archiwizacji)
  SELECT w.*, SYSTIMESTAMP
  FROM Wypozyczenia w
  WHERE w.Status = 'Zakonczone'
    AND CAST(w.Data_Zwrotu_Rzeczywista AS DATE) < v_data_graniczna;

  v_liczba_przeniesonych := SQL%ROWCOUNT;

  DELETE FROM Faktury
  WHERE ID_Wypozyczenia IN (
    SELECT ID_Wypozyczenia FROM Wypozyczenia
    WHERE Status = 'Zakonczone'
      AND CAST(Data_Zwrotu_Rzeczywista AS DATE) < v_data_graniczna
  );

  DELETE FROM Wypozyczenia
  WHERE Status = 'Zakonczone'
    AND CAST(Data_Zwrotu_Rzeczywista AS DATE) < v_data_graniczna;

  COMMIT;

  DBMS_OUTPUT.PUT_LINE('Zarchiwizowano: ' || v_liczba_przeniesonych || ' wypożyczeń');
  DBMS_OUTPUT.PUT_LINE('Data graniczna: ' || TO_CHAR(v_data_graniczna, 'DD-MM-YYYY'));
EXCEPTION
  WHEN OTHERS THEN
    ROLLBACK;
    RAISE;
END;
/

GRANT INSERT, UPDATE, DELETE ON Wypozyczenia_Archiwum TO db_datawriter;
GRANT EXECUTE ON ArchiwizujWypozyczenia TO db_procexecutor;
